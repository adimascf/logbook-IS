import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

df = pd.read_csv('distance_cds_after_IS.txt', sep='\t', header=None)
distance = df.iloc[:,-1] # last column

plt.figure(figsize=(10,6))
sns.histplot(distance, bins=100, color='skyblue', edgecolor='black')
plt.xlim(0, 15000)
plt.ylim(0, 17)
plt.xlabel('Distance (bp)')
plt.ylabel('Frequency')
plt.grid(axis='y', alpha=0.3)

plt.show()
